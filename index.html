async function importarCSV(event) { const arquivo = event.target.files[0]; if (!arquivo) return; const leitor = new FileReader(); leitor.onload = async function(e) { try { const conteudo = e.target.result; const linhas = conteudo.split(/\r\n|\n/).filter(l => l.trim() !== ''); if (linhas.length === 0) { alert('O arquivo CSV está vazio.'); return; } // Detecta automaticamente o separador predominante na primeira linha const primeiraLinha = linhas[0]; let separador = ';'; if ((primeiraLinha.match(/;/g) || []).length > 0) separador = ';'; else if ((primeiraLinha.match(/\t/g) || []).length > 0) separador = '\t'; else separador = ','; let novosLotes = []; let linhasIgnoradas = 0; linhas.forEach((linha, index) => { const colunas = linha.split(separador).map(c => c.replace(/^["']|["']$/g, '').trim()); // Pula linha de cabeçalho const col0Lower = (colunas[0] || '').toLowerCase(); const col3Lower = (colunas[3] || '').toLowerCase(); if (index === 0 && ( col0Lower.includes('forma') || col0Lower.includes('pagamento') || col3Lower.includes('data') )) { return; } if (colunas.length >= 6) { const formaPagamento = colunas[0] || 'Débito'; let parcelaCCBruta = (colunas[1] || '').replace(/^parcela\s*:\s*/i, '').trim(); // Tenta pegar a data da coluna 3 (D) ou coluna 2 (C) let dataStr = colunas[3] || ''; if (!dataStr.includes('/') && !dataStr.includes('-')) { dataStr = colunas[2] || ''; } const desc = colunas[4] || ''; const valStr = colunas[5] || ''; const subcat = colunas[6] || 'Geral'; const cat = colunas[7] || 'OUTROS'; const fixo = colunas[8] || ''; let val = parseValorExcel(valStr); if (val !== 0 && dataStr) { let dataFormatada = dataStr.trim(); if (dataFormatada.includes('-') && dataFormatada.length >= 10) { dataFormatada = dataFormatada.substring(0, 10); } else if (dataFormatada.includes('/')) { const partes = dataFormatada.split('/'); if (partes.length === 3) { const ano = partes[2].length === 2 ? `20${partes[2]}` : partes[2]; dataFormatada = `${ano}-${partes[1].padStart(2, '0')}-${partes[0].padStart(2, '0')}`; } } // Valida se a data ficou no formato YYYY-MM-DD if (/^\d{4}-\d{2}-\d{2}$/.test(dataFormatada)) { novosLotes.push({ forma_pagamento: formaPagamento, parcela_cc: parcelaCCBruta, data: dataFormatada, descricao: desc, valor: val, subcategoria: subcat, categoria: cat, fixo: fixo }); } else { linhasIgnoradas++; } } else { linhasIgnoradas++; } } else { linhasIgnoradas++; } }); if (novosLotes.length > 0) { const tamanhoLote = 500; for (let i = 0; i < novosLotes.length; i += tamanhoLote) { const lote = novosLotes.slice(i, i + tamanhoLote); const { error } = await supabaseClient.from('transacoes').insert(lote); if (error) { alert('Erro ao salvar no Supabase: ' + error.message); console.error('Detalhes do erro Supabase:', error); return; } } await carregarTransacoesDaNuvem(); alert(`${novosLotes.length} transações importadas com sucesso!${linhasIgnoradas > 0 ? ` (${linhasIgnoradas} linhas ignoradas)` : ''}`); } else { const exemploColunas = linhas[linhas.length > 1 ? 1 : 0].split(separador); alert( `Nenhuma transação válida foi encontrada no arquivo.\n\n` + `Separador detectado: "${separador}"\n` + `Colunas lidas na linha de exemplo (${exemploColunas.length} colunas):\n` + exemploColunas.map((c, idx) => `[${idx}]: ${c}`).join('\n') ); } } catch (erro) { alert('Erro ao processar o arquivo CSV: ' + erro.message); console.error(erro); } }; leitor.readAsText(arquivo, 'ISO-8859-1'); }
